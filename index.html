วิธีสร้าง Apps Script

1. เปิด Google Sheets ที่ต้องการ → Extensions → Apps Script

2. วางโค้ดด้านล่างทั้งหมด แทนที่โค้ดเดิม:

function doPost(e) {
  try {
    var ss = SpreadsheetApp.openById('1X4NBGK_8xokrgl3qf1F1XtCdUUhFeyKvXjQTaJMMkbI');
    var sheet = ss.getSheetByName('EmergencyCheck')
                || ss.insertSheet('EmergencyCheck');
    var data = JSON.parse(e.postData.contents);

    // Add header if sheet is empty
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        'วันที่','เวร','ผู้ตรวจสอบ','หัวหน้าหอ',
        'Laryngoscope(1)','ไฟ Lary1',
        'Laryngoscope(2)','ไฟ Lary2',
        'Emergency box 1','ยา box1',
        'Emergency box 2','ยา box2',
        'Defibrillator','Battery','Red dot',
        'หมายเหตุ','เวลาบันทึก'
      ]);
      sheet.getRange(1,1,1,17)
        .setBackground('#1A5276')
        .setFontColor('#FFFFFF')
        .setFontWeight('bold');
      sheet.setFrozenRows(1);
    }

    sheet.appendRow([
      data.date, data.shift, data.nurse, data.incharge,
      data.lary1, data.lary1_light,
      data.lary2, data.lary2_light,
      data.box1, data.box1_med,
      data.box2, data.box2_med,
      data.defib, data.battery, data.reddot,
      data.note, data.timestamp
    ]);

    return ContentService
      .createTextOutput(JSON.stringify({status:'ok'}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService
      .createTextOutput(JSON.stringify({status:'error',msg:err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  return ContentService
    .createTextOutput(JSON.stringify({status:'alive'}))
    .setMimeType(ContentService.MimeType.JSON);
}

// ── ดึง File ID รูปทั้งหมดจาก Drive Folder ──
function getFileIds() {
  var folderId = '1tfNKiu55ocZbXF3ovQaH7dz97tSOKsUM';
  var folder = DriveApp.getFolderById(folderId);
  var files = folder.getFiles();
  var ss = SpreadsheetApp.openById('1X4NBGK_8xokrgl3qf1F1XtCdUUhFeyKvXjQTaJMMkbI');
  var sheet = ss.getSheetByName('FileIDs') || ss.insertSheet('FileIDs');
  sheet.clearContents();
  sheet.appendRow(['FileName','FileID','ImgURL']);
  sheet.getRange(1,1,1,3)
    .setBackground('#1A5276').setFontColor('#fff').setFontWeight('bold');
  while (files.hasNext()) {
    var f = files.next();
    var id = f.getId();
    sheet.appendRow([
      f.getName(), id,
      'https://lh3.googleusercontent.com/d/' + id
    ]);
  }
  SpreadsheetApp.getUi()
    .alert('เสร็จแล้ว! ดู Sheet "FileIDs" ใน Spreadsheet');
}

3. กด Deploy → New deployment → Web app

4. Execute as: Me · Who has access: Anyone

5. คัดลอก Web App URL มาใส่ในช่องด้านบน แล้วกด "บันทึกการตั้งค่า"

🖼 วิธีดึงรูปภาพจริง (ทำครั้งเดียว)

6. ใน Apps Script เลือก function getFileIds → กด Run

7. เปิด Google Sheets จะเห็น Sheet ใหม่ชื่อ "FileIDs"

8. ส่ง Sheet นั้น (หรือ copy ข้อมูล) ให้ Claude → จะอัปเดต HTML ให้แสดงรูปจริงทันที
